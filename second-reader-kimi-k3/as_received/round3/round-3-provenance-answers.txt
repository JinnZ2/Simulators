import random

def planted_series(s, n):
    rng, x, sig = random.Random(s), 0.0, []
    for t in range(n):
        x = (0.05 + 0.9 * t / (n - 1)) * x + rng.gauss(0, 1)
        sig.append(x)
    return sig

def white_series(s, n):
    rng = random.Random(10000 + s)
    return [rng.gauss(0, 1) for _ in range(n)]

def make(kind, s, n):
    if kind == "planted":
        return planted_series(s, n)
    if kind == "white":
        return white_series(s, n)
    if kind == "drift":
        rng = random.Random(s)
        return [0.02 * t + rng.gauss(0, 1) for t in range(n)]
    if kind == "step":
        rng = random.Random(s)
        return [(3.0 if t >= n // 2 else 0.0) + rng.gauss(0, 1) for t in range(n)]
